import csv
from pathlib import Path
import matplotlib
matplotlib.use('Agg')
import matplotlib.pyplot as plt
root = Path('/home/kyberszittya/hymeko-core-amaterasu-20260905/clean/core/target/distributed-bench/results')
rows = list(csv.DictReader((root / 'platform-summary.csv').open()))
benchmarks = sorted({r['benchmark'] for r in rows})
platforms = [('windows', 'Windows native', '#176b9c'), ('native', 'Amaterasu native', '#ca641e'), ('arm64', 'Docker/QEMU ARM64', '#773b96')]
fig, ax = plt.subplots(figsize=(11, 6))
for group, (platform, label, color) in enumerate(platforms):
    selected = {r['benchmark']: r for r in rows if r['platform'] == platform}
    medians = [float(selected[b]['median_ns']) for b in benchmarks]
    low = [m - float(selected[b]['q1_ns']) for m, b in zip(medians, benchmarks)]
    high = [float(selected[b]['q3_ns']) - m for m, b in zip(medians, benchmarks)]
    y = [i + (group-1)*0.22 for i in range(len(benchmarks))]
    ax.errorbar(medians, y, xerr=[low, high], fmt='o', ms=4, capsize=3, color=color, label=label)
ax.set_yticks(range(len(benchmarks)))
ax.set_yticklabels([b.split('/', 1)[1] for b in benchmarks])
ax.invert_yaxis()
ax.set_xscale('log')
ax.set_xlabel('Nanoseconds per iteration-batch mean (log scale)')
ax.set_title('Core admission and replay: identical source and dependency lock', loc='left', pad=34)
ax.text(0, 1.035, '3 process runs/platform; dots = pooled median, whiskers = IQR; ARM64 is emulated', transform=ax.transAxes, fontsize=9)
ax.grid(axis='x', which='both', alpha=0.2)
ax.legend(loc='lower right', frameon=False)
fig.tight_layout()
for suffix in ['png', 'svg', 'pdf']:
    fig.savefig(root / ('platform-latency.'+suffix), dpi=180)
print('matplotlib', matplotlib.__version__)
