# One campaign transcript, executed with python3 on Amaterasu; not a new runner API.
import json
import os
from pathlib import Path
import subprocess
import time

root = Path('/home/kyberszittya/hymeko-core-amaterasu-20260905/clean/core')
work = root / 'target/distributed-bench'
results = work / 'results'
results.mkdir(exist_ok=True)
pattern = '^(ir_stream_admission/|ir_stream_replay/(exact_duplicate|conflicting_duplicate|replay_window_miss)$)'
runtime = 'sha256:32d322b19846336d25f755f73618a448e3621982d52c48064e95af8b3dcbc2d9'
environment = dict(os.environ, RAYON_NUM_THREADS='4', LC_ALL='C')

def execute(label, command, container=None, extra_env=None):
    folder = results / label
    folder.mkdir(exist_ok=True)
    assert not (folder / 'process.json').exists(), 'Never overwrite a finished run'
    started = time.time()
    peak_rss_kib = 0
    peak_cgroup_bytes = 0
    guest_pid = None
    cgroup = None
    with (folder / 'stdout.log').open('wb') as stdout, (folder / 'stderr.log').open('wb') as stderr:
        process = subprocess.Popen(command, cwd=work, env=extra_env or environment, stdout=stdout, stderr=stderr)
        next_progress = 10
        while process.poll() is None:
            if container and guest_pid is None:
                probe = subprocess.run(['docker', 'inspect', '--format', '{{.State.Pid}}', container], capture_output=True)
                if probe.returncode == 0 and int(probe.stdout or b'0') > 0:
                    guest_pid = int(probe.stdout)
                    try:
                        path = Path(f'/proc/{guest_pid}/cgroup').read_text().strip().split('::', 1)[1]
                        cgroup = Path('/sys/fs/cgroup') / path.lstrip('/')
                    except FileNotFoundError:
                        pass
            if guest_pid:
                try:
                    status = Path(f'/proc/{guest_pid}/status').read_text()
                    for line in status.splitlines():
                        if line.startswith('VmHWM:'):
                            peak_rss_kib = max(peak_rss_kib, int(line.split()[1]))
                    if cgroup:
                        peak_cgroup_bytes = max(peak_cgroup_bytes, int((cgroup / 'memory.peak').read_text()))
                except (FileNotFoundError, ProcessLookupError):
                    pass
            elapsed = time.time() - started
            if elapsed >= next_progress:
                print(f'PROGRESS {label}: {elapsed:.1f}s, QEMU VmHWM {peak_rss_kib} KiB', flush=True)
                next_progress += 10
            time.sleep(0.1)
    metrics = dict(command=command, exit=process.returncode, started_utc_epoch=started,
                   wall_seconds=time.time()-started, sampled_qemu_peak_rss_kib=peak_rss_kib,
                   sampled_cgroup_peak_bytes=peak_cgroup_bytes, poll_seconds=0.1)
    if container:
        info = json.loads(subprocess.check_output(['docker', 'inspect', container]))[0]
        metrics['container'] = dict(state=info['State'], image=info['Image'],
                                   memory_limit=info['HostConfig']['Memory'],
                                   memory_swap=info['HostConfig']['MemorySwap'],
                                   nano_cpus=info['HostConfig']['NanoCpus'])
        subprocess.run(['docker', 'rm', container], check=True, stdout=subprocess.DEVNULL)
    (folder / 'process.json').write_text(json.dumps(metrics, indent=2)+'\n')
    print(f'DONE {label}: exit={process.returncode}, wall={metrics["wall_seconds"]:.3f}s', flush=True)
    if process.returncode:
        print((folder / 'stderr.log').read_text(), flush=True)
        print((folder / 'stdout.log').read_text(), flush=True)
        raise SystemExit(process.returncode)

def native(label, executable, arguments):
    folder = results / label
    folder.mkdir(exist_ok=True)
    env = dict(environment, CRITERION_HOME=str(folder / 'criterion'))
    command = ['systemd-run', '--user', '--scope', '--quiet', '-p', 'MemoryMax=16G',
               '-p', 'MemorySwapMax=0', '-p', 'CPUQuota=400%', '/usr/bin/time', '-v',
               '-o', str(folder / 'time.txt'), str(work / executable), *arguments]
    execute(label, command, extra_env=env)

def arm(label, executable, arguments):
    folder = results / label
    folder.mkdir(exist_ok=True)
    container = 'hymeko-core-distributed-' + label + '-20260906'
    command = ['docker', 'run', '--name', container, '--platform', 'linux/arm64',
               '--network=none', '--memory=16g', '--memory-swap=16g', '--cpus=4',
               '-e', 'RAYON_NUM_THREADS=4', '-e', 'LC_ALL=C', '-e', 'CRITERION_HOME=/results/criterion',
               '-v', str(work / 'qemu-aarch64-static')+':/qemu:ro',
               '-v', str(work / executable)+':/program:ro', '-v', str(folder)+':/results',
               '--entrypoint', '/qemu', runtime, '/program', *arguments]
    execute(label, command, container=container)

inventory = {
    'source': subprocess.check_output(['git', '-C', str(root), 'rev-parse', 'HEAD']).decode().strip(),
    'uname': subprocess.check_output(['uname', '-a']).decode().strip(),
    'lscpu': subprocess.check_output(['lscpu']).decode(),
    'qemu': subprocess.check_output([str(work / 'qemu-aarch64-static'), '--version']).decode(),
    'runtime': json.loads(subprocess.check_output(['docker', 'image', 'inspect', runtime]))[0],
    'initial_load': Path('/proc/loadavg').read_text().strip(),
    'binfmt_handlers': sorted(p.name for p in Path('/proc/sys/fs/binfmt_misc').iterdir()),
}
(results / 'inventory.json').write_text(json.dumps(inventory, indent=2)+'\n')
assert inventory['runtime']['Architecture'] == 'arm64'
native('native-correctness', 'native-tests', ['ir_stream_admission', '--test-threads=4'])
arm('arm64-correctness', 'arm64-tests', ['ir_stream_admission', '--test-threads=4'])
native('native-smoke', 'native-bench', ['--test', pattern])
arm('arm64-smoke', 'arm64-bench', ['--test', pattern])
for index in range(1, 4):
    native(f'native-run{index}', 'native-bench', ['--bench', pattern])
for index in range(1, 4):
    arm(f'arm64-run{index}', 'arm64-bench', ['--bench', pattern])
